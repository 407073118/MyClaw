# Quick 260508-ic3 — Plan

## 设计

「定时任务执行就是对话」—— 抽屉做不了"接着聊"，必须独立页。详情页布局：

```
┌─ /time/jobs/:id ─────────────────────────────────────┐
│ ← 日程规划    每日热点新闻收集 [Prompt]              │
│              每天 09:00 · 上次成功 5/8 11:06        │
│              [立即执行] [编辑] [暂停] [删除]          │
├──────────────────────────────────────────────────────┤
│ 💬 对话                                              │
│                                                       │
│ [user, 5/8 11:06]      用 5 个要点总结今天的科技…    │
│ [assistant, 5/8 11:06] # 1. … ## 2. … (markdown)   │
│ [user, 5/9 09:00]      接着上一次再追问…             │
│ [assistant, 5/9 09:00] …                              │
│                                                       │
│ ┌──────────────────────────────────────────┐ [发送] │
│ │ 输入消息继续对话…                         │       │
│ └──────────────────────────────────────────┘        │
├──────────────────────────────────────────────────────┤
│ ⚡ 执行记录 (3)                                       │
│ ✓ 5/9 09:00 → 09:00  succeeded  "2 句要点摘要…"     │
│ ✓ 5/8 11:06 → 11:06  succeeded  "今天的热点是…"     │
│ ✗ 5/7 11:06           failed   "fetch failed"       │
└──────────────────────────────────────────────────────┘
```

### Task 1 — 新建 TimeJobDetailPage

**files:** `desktop/src/renderer/pages/TimeJobDetailPage.tsx` (new)

**action:** 单文件实现整个详情页：

```tsx
import React, { useEffect, useMemo, useRef, useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
import { useWorkspaceStore } from "../stores/workspace";
import MarkdownView from "../components/MarkdownView";
import { formatJobFrequency } from "../utils/frequency";
import { textOfContent } from "@shared/contracts";
import type { ChatMessage, ExecutionRun, ScheduleJob, ScheduleJobExecutor } from "@shared/contracts";

export default function TimeJobDetailPage() {
  const { id } = useParams<{ id: string }>();
  const navigate = useNavigate();
  const job = useWorkspaceStore((s) => s.time.scheduleJobs.find((j) => j.id === id));
  const sessions = useWorkspaceStore((s) => s.sessions);
  const allRuns = useWorkspaceStore((s) => s.time.executionRuns);
  const siliconPersons = useWorkspaceStore((s) => s.siliconPersons);

  const session = useMemo(
    () => (job?.sessionId ? sessions.find((s) => s.id === job.sessionId) ?? null : null),
    [job?.sessionId, sessions],
  );
  const runs = useMemo(
    () => allRuns.filter((r) => r.jobId === id).sort((a, b) => b.startedAt.localeCompare(a.startedAt)),
    [allRuns, id],
  );
  const visibleMessages = useMemo(() => filterDisplayableMessages(session?.messages ?? []), [session?.messages]);

  const [draft, setDraft] = useState("");
  const [sending, setSending] = useState(false);
  const [running, setRunning] = useState(false);
  const messagesEndRef = useRef<HTMLDivElement | null>(null);
  const chatScrollRef = useRef<HTMLDivElement | null>(null);

  // 进入页面时自动选中该 session（让 sendMessage 能命中）
  useEffect(() => {
    if (job?.sessionId) {
      useWorkspaceStore.getState().selectSession(job.sessionId);
    }
  }, [job?.sessionId]);

  // 消息变化时滚到底部
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
  }, [visibleMessages.length]);

  // 进入后刷一次 execution runs（确保最新）
  useEffect(() => {
    void useWorkspaceStore.getState().refreshExecutionRuns();
  }, [id]);

  if (!job) {
    return (
      <main className="time-job-detail-page">
        <header className="time-job-detail__header">
          <button type="button" className="time-job-detail__back" onClick={() => navigate("/time")}>← 日程规划</button>
        </header>
        <div className="time-job-detail__empty">
          <p>找不到这个定时任务，可能已删除。</p>
        </div>
      </main>
    );
  }

  // 从 messages.length 切片 + sortedRuns 中提取 lastSummary 用于 header
  // ...
  
  async function handleRunNow() { setRunning(true); try { await store.executeScheduleJobNow(job.id); } finally { setRunning(false); } }
  function handleEdit() { void navigate("/time", { state: { editJobId: job.id } }); }
  async function handleTogglePause() { await store.updateScheduleJob({ ...job, status: job.status === "paused" ? "scheduled" : "paused" }); }
  async function handleDelete() {
    if (!confirm(`确认删除「${job.title}」？`)) return;
    await store.deleteScheduleJob(job.id);
    void navigate("/time");
  }
  async function handleSend(event) {
    event.preventDefault();
    const trimmed = draft.trim();
    if (!trimmed || sending || !session) return;
    setSending(true);
    try {
      const ws = useWorkspaceStore.getState();
      ws.selectSession(session.id);
      await ws.sendMessage(trimmed);
      setDraft("");
    } finally {
      setSending(false);
    }
  }

  return (
    <main className="time-job-detail-page" data-testid="time-job-detail">
      <header className="time-job-detail__header">
        <button onClick={() => navigate("/time")} className="time-job-detail__back">← 日程规划</button>
        <div className="time-job-detail__heading">
          <div className="time-job-detail__title-row">
            <h1>{job.title}</h1>
            <span className={`job-type-chip job-type-chip--${job.executor}`}>{formatExecutorLabel(job.executor)}</span>
            <span className={`status-badge status-badge--${...}`}>{...}</span>
          </div>
          <p className="time-job-detail__meta">
            {formatJobFrequency(job, (iso) => formatDateTime(iso, job.timezone))}
            {job.description && job.executor !== "assistant_prompt" ? ` · ${job.description}` : ""}
          </p>
        </div>
        <div className="time-job-detail__actions">
          <button onClick={handleRunNow} disabled={running}>{running ? "执行中…" : "立即执行"}</button>
          <button onClick={handleEdit}>编辑</button>
          <button onClick={handleTogglePause}>{job.status === "paused" ? "恢复" : "暂停"}</button>
          <button onClick={handleDelete} className="is-danger">删除</button>
        </div>
      </header>

      {job.executor === "assistant_prompt" && session ? (
        <section className="time-job-detail__chat">
          <header><h2>对话</h2><span>{visibleMessages.length} 条消息</span></header>
          <div ref={chatScrollRef} className="time-job-detail__messages">
            {visibleMessages.length === 0 ? (
              <div className="time-job-detail__chat-empty">
                <p>这个任务还没产生过对话。</p>
                <p>到点触发或点「立即执行」会生成第一条消息。</p>
              </div>
            ) : (
              visibleMessages.map((message) => <ChatBubble key={message.id} message={message} />)
            )}
            <div ref={messagesEndRef} />
          </div>
          <form className="time-job-detail__compose" onSubmit={handleSend}>
            <textarea
              value={draft}
              onChange={(e) => setDraft(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === "Enter" && (e.metaKey || e.ctrlKey)) {
                  e.preventDefault();
                  void handleSend(e);
                }
              }}
              placeholder="继续往下聊…（Cmd / Ctrl + Enter 发送）"
              rows={3}
              disabled={sending}
            />
            <button type="submit" disabled={sending || !draft.trim()}>{sending ? "发送中…" : "发送"}</button>
          </form>
        </section>
      ) : null}

      <section className="time-job-detail__runs">
        <header>
          <h2>执行记录</h2>
          <span>{runs.length} 次</span>
        </header>
        {runs.length === 0 ? (
          <p className="time-job-detail__empty-hint">暂无执行记录</p>
        ) : (
          <ol className="time-job-detail__run-list">
            {runs.map((run) => <RunRow key={run.id} run={run} timezone={job.timezone} />)}
          </ol>
        )}
      </section>

      <style>{styles}</style>
    </main>
  );
}

function ChatBubble({ message }: { message: ChatMessage }) {
  if (message.role === "system") return null;
  if (message.role === "tool") return <ToolBubble message={message} />;
  return (
    <div className={`bubble bubble--${message.role}`}>
      <div className="bubble__time">{formatTime(message.createdAt)}</div>
      <div className="bubble__content">
        <MarkdownView source={textOfContent(message.content)} className="bubble__markdown" />
      </div>
    </div>
  );
}

function ToolBubble({ message }: { message: ChatMessage }) {
  return (
    <div className="bubble bubble--tool">
      <div className="bubble__time">工具结果 · {formatTime(message.createdAt)}</div>
      <details className="bubble__tool-details">
        <summary>展开查看</summary>
        <pre>{textOfContent(message.content)}</pre>
      </details>
    </div>
  );
}

function RunRow({ run, timezone }: { run: ExecutionRun; timezone: string }) { /* row UI */ }

function filterDisplayableMessages(messages: ChatMessage[]): ChatMessage[] {
  return messages.filter((m) => m.role !== "system");
}
```

CSS 设计要点：
- 整体采用 `--bg-base` 页面 + 上中下三段
- header 用 `.glass-card` + radius-xl
- 对话区 messages 容器固定高度 + overflow-y auto，bubbles 间距 16
- user bubble 右对齐 cyan 边框，assistant bubble 左对齐 default 边框
- compose textarea 跟随 ChatPage 风格但更紧凑
- 执行记录列表用 `.execution-history-row` 同款 token 但对齐网格化（避免 drawer 那种"卡片堆"感）
