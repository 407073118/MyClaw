    import type { CodeNode, DocumentIR, HeadingNode, InlineRun, ListNode, OutlineItem, ParagraphNode, QuoteNode, TableNode } from "@shared/contracts";
    import type { DocumentParser, ParseInput } from "../parser-registry";

    export async function parseMarkdownBuffer(input: ParseInput): Promise<DocumentIR> {
      let marked: any;
      try {
        marked = require("marked");
      } catch {
        throw new Error("[E_DOC_DEP_MISSING] marked not installed.");
      }
      const text = input.buffer.toString("utf8");
      const tokens = marked.lexer ? marked.lexer(text) : marked.marked.lexer(text);
      const body: DocumentIR["body"] = [];
      const outline: OutlineItem[] = [];
      let lastHeading: string | undefined;

      for (const tk of tokens) {
        switch (tk.type) {
          case "heading": {
            const level = Math.min(6, Math.max(1, tk.depth)) as 1 | 2 | 3 | 4 | 5 | 6;
            const node: HeadingNode = { kind: "heading", level, runs: [{ text: tk.text }], locator: { heading: tk.text } };
            body.push(node);
            outline.push({ level, title: tk.text, locator: { heading: tk.text } });
            lastHeading = tk.text;
            break;
          }
          case "paragraph": {
            const node: ParagraphNode = { kind: "paragraph", runs: [{ text: tk.text }], locator: { heading: lastHeading } };
            body.push(node);
            break;
          }
          case "list": {
            const items: InlineRun[][] = (tk.items ?? []).map((it: any) => [{ text: String(it.text ?? "") }]);
            const node: ListNode = { kind: "list", ordered: !!tk.ordered, items, locator: { heading: lastHeading } };
            body.push(node);
            break;
          }
          case "code": {
            const node: CodeNode = { kind: "code", lang: tk.lang || undefined, text: tk.text ?? "", locator: { heading: lastHeading } };
            body.push(node);
            break;
          }
          case "table": {
            const headerRow: InlineRun[][] = (tk.header ?? []).map((cell: any) => [{ text: String(cell.text ?? cell ?? "") }]);
            const dataRows: InlineRun[][][] = (tk.rows ?? []).map((row: any[]) => row.map((cell: any) => [{ text: String(cell.text ?? cell ?? "") }]));
            const rows: InlineRun[][][] = [headerRow, ...dataRows];
            const node: TableNode = { kind: "table", rows, locator: { heading: lastHeading } };
            body.push(node);
            break;
          }
          case "blockquote": {
            const node: QuoteNode = { kind: "quote", runs: [{ text: tk.text ?? "" }], locator: { heading: lastHeading } };
            body.push(node);
            break;
          }
          // skip space / html / hr — no semantic value in IR
        }
      }

      return {
        source: { path: input.path, format: "md", bytes: input.buffer.length, sha256: input.sha256 },
        meta: { words: Math.round(text.length / 5) },
        outline,
        body,
        media: [],
      };
    }

    export async function parsePlainTextBuffer(input: ParseInput): Promise<DocumentIR> {
      // Strip BOM, split on \n\n+, emit ParagraphNode per block
      let text = input.buffer.toString("utf8");
      if (text.charCodeAt(0) === 0xFEFF) text = text.slice(1);
      const blocks = text.split(/\n{2,}/).map((b) => b.trim()).filter((b) => b.length > 0);
      const body: ParagraphNode[] = blocks.map<ParagraphNode>((b) => ({
        kind: "paragraph",
        runs: [{ text: b }],
        locator: {},
      }));
      return {
        source: { path: input.path, format: "txt", bytes: input.buffer.length, sha256: input.sha256 },
        meta: { words: Math.round(text.length / 5) },
        outline: [],
        body,
        media: [],
      };
    }

    export const mdParser: DocumentParser = { format: "md", parse: parseMarkdownBuffer };
    export const txtParser: DocumentParser = { format: "txt", parse: parsePlainTextBuffer };
    ```

    import type { DocumentIR, InlineRun, OutlineItem, SheetNode, TableNode } from "@shared/contracts";
    import type { DocumentParser, ParseInput } from "../parser-registry";

    const CANDIDATE_DELIMITERS = [",", "\t", ";", "|"] as const;

    export function detectDelimiter(sample: string): string {
      const lines = sample.split(/\r?\n/).filter((l) => l.length > 0).slice(0, 5);
      if (lines.length === 0) return ",";
      let best = ",";
      let bestConsistency = -1;
      for (const d of CANDIDATE_DELIMITERS) {
        const counts = lines.map((l) => (l.match(new RegExp(`\\${d}`, "g")) ?? []).length);
        const first = counts[0];
        if (first === 0) continue;
        const consistent = counts.every((c) => c === first);
        if (consistent && first > bestConsistency) {
          bestConsistency = first;
          best = d;
        }
      }
      return best;
    }

    /** Robust line-level CSV parser honoring quoted fields and escaped double-quotes. */
    export function parseCsvText(text: string, delim: string): string[][] {
      const rows: string[][] = [];
      let cur: string[] = [];
      let field = "";
      let inQuotes = false;
      for (let i = 0; i < text.length; i++) {
        const ch = text[i];
        if (inQuotes) {
          if (ch === "\"" && text[i + 1] === "\"") { field += "\""; i++; continue; }
          if (ch === "\"") { inQuotes = false; continue; }
          field += ch;
        } else {
          if (ch === "\"") { inQuotes = true; continue; }
          if (ch === delim) { cur.push(field); field = ""; continue; }
          if (ch === "\n") { cur.push(field); rows.push(cur); cur = []; field = ""; continue; }
          if (ch === "\r") { continue; }
          field += ch;
        }
      }
      if (field.length > 0 || cur.length > 0) { cur.push(field); rows.push(cur); }
      return rows;
    }

    export async function parseCsvBuffer(input: ParseInput): Promise<DocumentIR> {
      let text = input.buffer.toString("utf8");
      if (text.charCodeAt(0) === 0xFEFF) text = text.slice(1);
      const delim = detectDelimiter(text.slice(0, 4096));
      const rows = parseCsvText(text, delim);
      const cols = rows.reduce((acc, r) => Math.max(acc, r.length), 0);
      const tableRows: InlineRun[][][] = rows.map((r) => {
        const padded = [...r];
        while (padded.length < cols) padded.push("");
        return padded.map((c) => [{ text: c }]);
      });
      const table: TableNode = { kind: "table", rows: tableRows, locator: { sheet: "csv" } };
      const sheet: SheetNode = { kind: "sheet", name: "csv", dims: { rows: rows.length, cols }, table, locator: { sheet: "csv" } };
      const outline: OutlineItem[] = [{ level: 1, title: "csv", locator: { sheet: "csv" } }];
      return {
        source: { path: input.path, format: "csv", bytes: input.buffer.length, sha256: input.sha256 },
        meta: {},
        outline,
        body: [sheet],
        media: [],
      };
    }

    export const csvParser: DocumentParser = { format: "csv", parse: parseCsvBuffer };
    ```
