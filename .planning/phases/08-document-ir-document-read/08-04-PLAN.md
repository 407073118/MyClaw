Migrate the existing `xlsx.extract` behavior onto the new DocumentIR pipeline while keeping the legacy `xlsx.extract` toolId as a user-facing alias that shells to `document.read` internally.

Purpose: The original `xlsx.extract` works (line 1303-1372 of builtin-tool-executor.ts) but produces flat Markdown only. Wrapping it through IR gives the same output for legacy callers AND enables `document.read mode=stats/outline/search` for xlsx from the same parse.

Constraint: backward compatibility is non-negotiable. The `xlsx_extract` schema and `xlsx.extract` dispatch must keep producing Markdown tables byte-for-byte compatible with the pre-migration behavior (first sheet, ≤ maxRows, pipe-escaped, truncation footer).

Zero-Python, zero new dependencies (xlsx package already in package.json).
</objective>

    ```ts
    import { extname } from "node:path";
    import type { DocumentFormat, DocumentIR, InlineRun, OutlineItem, SheetNode, TableNode } from "@shared/contracts";
    import type { DocumentParser, ParseInput } from "../parser-registry";

    export async function parseXlsxBuffer(input: ParseInput): Promise<DocumentIR> {
      // Lazy require so main-process startup doesn't hard-fail if xlsx is absent
      let xlsxMod: any;
      try {
        xlsxMod = require("xlsx");
      } catch {
        throw new Error("[E_DOC_DEP_MISSING] xlsx package not installed. Run `pnpm install` in desktop/.");
      }

      const wb = xlsxMod.read(input.buffer, { type: "buffer" });
      const sheetNames: string[] = wb.SheetNames ?? [];
      const body: SheetNode[] = [];
      const outline: OutlineItem[] = [];

      for (let i = 0; i < sheetNames.length; i++) {
        const name = sheetNames[i];
        const sheet = wb.Sheets[name];
        const aoa: unknown[][] = xlsxMod.utils.sheet_to_json(sheet, { header: 1, defval: "" });

        // Expand merged cells in-place
        const merges = sheet["!merges"] as Array<{ s: { r: number; c: number }; e: { r: number; c: number } }> | undefined;
        if (merges) {
          for (const m of merges) {
            const anchor = aoa[m.s.r]?.[m.s.c] ?? "";
            for (let r = m.s.r; r <= m.e.r; r++) {
              if (!aoa[r]) aoa[r] = [];
              for (let c = m.s.c; c <= m.e.c; c++) {
                aoa[r][c] = anchor;
              }
            }
          }
        }

        const rows = aoa.length;
        const cols = aoa.reduce((acc, r) => Math.max(acc, r.length), 0);
        const tableRows: InlineRun[][][] = aoa.map((r) => {
          const padded: unknown[] = [...r];
          while (padded.length < cols) padded.push("");
          return padded.map((cell) => [{ text: String(cell ?? "") } as InlineRun]);
        });
        const table: TableNode = { kind: "table", rows: tableRows, locator: { sheet: name } };
        body.push({ kind: "sheet", name, dims: { rows, cols }, table, locator: { sheet: name } });
        outline.push({ level: 1, title: name, locator: { sheet: name } });
      }

      const ext = extname(input.path).toLowerCase().replace(".", "");
      const format: DocumentFormat = (ext === "xlsx" || ext === "xls" || ext === "xlsm") ? (ext as DocumentFormat) : "xlsx";

      return {
        source: { path: input.path, format, bytes: input.buffer.length, sha256: input.sha256 },
        meta: {},
        outline,
        body,
        media: [],
      };
    }

    export const xlsxParser: DocumentParser = {
      format: "xlsx",
      parse: parseXlsxBuffer,
    };

    export const xlsParser: DocumentParser = { format: "xls", parse: parseXlsxBuffer };
    export const xlsmParser: DocumentParser = { format: "xlsm", parse: parseXlsxBuffer };
    ```
