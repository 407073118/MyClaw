       ```ts
       import type { DocumentIR, InlineRun, OutlineItem, PageBreakNode, ParagraphNode } from "@shared/contracts";
       import type { DocumentParser, ParseInput } from "../parser-registry";

       export async function parsePdfBuffer(input: ParseInput): Promise<DocumentIR> {
         let pdfjs: any;
         try {
           pdfjs = require("pdfjs-dist/legacy/build/pdf.js");
         } catch {
           throw new Error("[E_DOC_DEP_MISSING] pdfjs-dist not installed. Run `pnpm install` in desktop/.");
         }

         const loadingTask = pdfjs.getDocument({
           data: new Uint8Array(input.buffer),
           useSystemFonts: true,
           isEvalSupported: false,       // disable PDF.js eval
           disableFontFace: true,
         });
         const doc = await loadingTask.promise;
         const numPages: number = doc.numPages;
         const body: Array<PageBreakNode | ParagraphNode> = [];
         const outline: OutlineItem[] = [];

         for (let p = 1; p <= numPages; p++) {
           const page = await doc.getPage(p);
           const content = await page.getTextContent();
           const items: Array<{ str: string }> = content.items ?? [];
           body.push({ kind: "pageBreak", page: p, locator: { page: p } });
           outline.push({ level: 1, title: `Page ${p}`, locator: { page: p } });

           if (items.length === 0 || items.every((it) => !it.str || it.str.trim() === "")) {
             const runs: InlineRun[] = [{ text: "(scanned page — no extractable text)" }];
             body.push({ kind: "paragraph", runs, locator: { page: p } });
             continue;
           }
           // naive paragraph reconstruction: join items, split on double-newline or large y-gap (simpler: join items, then split on sentence boundaries)
           const text = items.map((it) => it.str).join(" ").replace(/\s+/g, " ").trim();
           // One paragraph per page — adequate for mode=read; sub-paragraph boundaries can be refined later
           const runs: InlineRun[] = [{ text }];
           body.push({ kind: "paragraph", runs, locator: { page: p } });
         }

         return {
           source: { path: input.path, format: "pdf", bytes: input.buffer.length, sha256: input.sha256 },
           meta: { pages: numPages },
           outline,
           body,
           media: [],
         };
       }

       export const pdfParser: DocumentParser = { format: "pdf", parse: parsePdfBuffer };
       ```
