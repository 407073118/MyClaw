      getRoot(): string;

      enforceLru(): Promise<void>;
    };

    export function createDocCache(opts: DocCacheOptions): DocCache {
      const root = join(opts.rootDir, "docCache");
      const maxBytes = opts.maxBytes ?? DEFAULT_CACHE_MAX_BYTES;

    }

    /** Compute sha256 of a buffer (hex, lower-case). */
    export function sha256OfBuffer(buf: Buffer): string {
      return createHash("sha256").update(buf).digest("hex");

    Implementation rules:
    - `getOrBuild(sha, builder)`:
      1. Compute `entryDir = <root>/docCache/<sha>`
      2. If `ir.json` exists and parses, read meta.json, increment hits, set lastAccess=now, write meta.json, return parsed IR
      3. If missing or `JSON.parse` fails, ensure entryDir+mediaDir exist, invoke `await builder(mediaDir)`, write ir.json, compute total size via `recursiveSize(entryDir)`, write meta.json `{ sha256, cachedAt: now, lastAccess: now, hits: 0, bytes }`
      4. After writing, call `enforceLru()` (best-effort; log errors, don't throw)
      5. Log at info: `log.info("[doc-cache] miss -> built", { sha256: sha, bytes })` or `log.info("[doc-cache] hit", { sha256: sha, hits })`
    - `enforceLru()`:
      1. `readdir(root)` to list sha directories; for each read meta.json
      2. Sum bytes; if ≤ maxBytes return
      3. Sort entries ascending by `lastAccess`
      4. Remove entries (rm -rf their dir) until under cap; log each eviction
    - `clear()` = `rm -rf <root>`.
    - All disk IO uses `node:fs/promises`. Handle ENOENT by treating as miss.

For LRU test, set `maxBytes: 1024` and write a few dummy entries, assert smallest-lastAccess entries removed first.
