    ```ts
    import { existsSync } from "node:fs";
    import { readFile, stat } from "node:fs/promises";

    import type { DocumentFormat, DocumentIR, Locator } from "@shared/contracts";
    import { createLogger } from "../logger";
    import type { DocCache } from "./doc-cache";
    import { sha256OfBuffer } from "./doc-cache";
    import { getParser } from "./parser-registry";
    import { renderIrToMarkdown } from "./ir-to-markdown";

    const log = createLogger("document-read");

    export const DOC_MAX_BYTES = 50 * 1024 * 1024;
    export const DOC_MAX_CHARS_HARD_CAP = 32000;
    export const DOC_DEFAULT_MAX_CHARS = 8000;

    export type DocumentReadMode = "stats" | "outline" | "read" | "search";

    export type DocumentReadArgs = {
      path: string;
      mode: DocumentReadMode;
      locator?: Locator;
      query?: string;
      maxChars?: number;
      format?: "markdown" | "json";
      includeImages?: "none" | "refs" | "inline";
    };

    export type DocumentReadDeps = {
      cache: DocCache;
      /** Caller supplies an already-resolved absolute path (PathAccessPolicy already ran). */
      resolvedPath: string;
      sessionId?: string | null;
    };

    export type DocumentReadResult =
      | { success: true; output: string }
      | { success: false; output: ""; error: string };

    export async function executeDocumentRead(args: DocumentReadArgs, deps: DocumentReadDeps): Promise<DocumentReadResult> {
      // 1. Size gate (50MB)
      // 2. Format detection from extname; map to DocumentFormat; reject unsupported (esp. ".doc")
      // 3. Read buffer + compute sha256; cache.getOrBuild(sha, mediaDir => parser.parse({ path, buffer, sha256, mediaDir }))
      // 4. Route by mode:
      //    - stats: JSON.stringify({ format, bytes, sha256, meta, outlineCount, bodyCount })
      //    - outline: render outline as "- Lv{level}: title (page=N/slide=N/heading=X)"
      //    - read: slice body by locator (heading | page | slide | sheet | range), then renderIrToMarkdown
      //    - search: scan body for runs containing query (case-insensitive substring), return first 20 with locator line prefix
      // 5. Clamp maxChars: Math.min(Math.max(args.maxChars ?? 8000, 100), 32000)
      // 6. Log: log.info("read done", { sha256, mode, path, returnedBytes: output.length })
      // 7. Map thrown parser errors to { success: false, error: `[E_DOC_PARSE_FAILED] ...` }
    }

    /** Map file extension to DocumentFormat; returns null for unsupported (including legacy .doc). */
    export function detectFormat(path: string): DocumentFormat | null {
      const ext = extname(path).toLowerCase();
      switch (ext) {
        case ".xlsx": return "xlsx";
        case ".xls": return "xls";
        case ".xlsm": return "xlsm";
        case ".docx": return "docx";
        case ".pdf": return "pdf";
        case ".pptx": return "pptx";
        case ".md": return "md";
        case ".txt": return "txt";
        case ".csv": return "csv";
        default: return null; // includes .doc -> guidance surfaced to caller
      }
    }
    ```

    - `searchBody(body, query, limit=20)` — descend into slide bodies & notes too; for each node whose runs concat contains `query.toLowerCase()`, emit a result.
