    ```ts
    import type { DocumentFormat, DocumentIR } from "@shared/contracts";

    export type ParseInput = {
      /** Absolute filesystem path already resolved by PathAccessPolicy. */
      path: string;
      /** Raw file buffer. */
      buffer: Buffer;
      /** Pre-computed sha256 of `buffer` (lower-case hex). */
      sha256: string;
      /** Where the parser may write extracted media; directory already exists. */
      mediaDir: string;
    };

    export interface DocumentParser {
      readonly format: DocumentFormat;
      parse(input: ParseInput): Promise<DocumentIR>;
    }

    const registry = new Map<DocumentFormat, DocumentParser>();

    export function registerParser(parser: DocumentParser): void {
      registry.set(parser.format, parser);
    }

    export function getParser(format: DocumentFormat): DocumentParser | null {
      return registry.get(format) ?? null;
    }

    export function listRegisteredFormats(): DocumentFormat[] {
      return Array.from(registry.keys());
    }

    /** Test-only reset; never call from production code. */
    export function __resetParserRegistryForTests(): void {
      registry.clear();
    }
    ```

    File `desktop/src/main/services/document/ir-to-markdown.ts`:
    ```ts
    import type { DocumentIR, DocumentNode, InlineRun } from "@shared/contracts";

    export type RenderOptions = {
      maxChars?: number;           // default: no limit at this layer
      includeImages?: "none" | "refs" | "inline"; // default "refs"
    };

    export function renderIrToMarkdown(ir: DocumentIR, opts: RenderOptions = {}): string {
      const includeImages = opts.includeImages ?? "refs";
      const out: string[] = [];
      for (const node of ir.body) {
        out.push(renderNode(node, includeImages));
      }
      const full = out.join("").trimEnd() + "\n";
      if (typeof opts.maxChars === "number" && full.length > opts.maxChars) {
        return full.slice(0, opts.maxChars) + "\n\n...（已截断）";
      }
      return full;
    }

    // renderNode, renderRuns, renderTable, renderSlide, renderSheet, renderList helpers
    ```
