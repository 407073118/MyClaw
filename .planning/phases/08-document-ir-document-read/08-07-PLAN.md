    ```ts
    import type { DocumentIR, InlineRun, OutlineItem, ParagraphNode, SlideNode } from "@shared/contracts";
    import type { DocumentParser, ParseInput } from "../parser-registry";

    const XXE_MARKER = "<!DOCTYPE";

    function assertNoDoctype(xml: string): void {
      if (xml.includes(XXE_MARKER)) {
        throw new Error(`[E_DOC_XXE_BLOCKED] External entity declaration detected in pptx part.`);
      }
    }

    /** Extract all <a:t>...</a:t> text, grouping by <a:p> paragraphs. Returns an array of paragraphs, each a string. */
    function extractPptxParagraphs(xml: string): string[] {
      assertNoDoctype(xml);
      const paragraphs: string[] = [];
      const pRegex = /<a:p\b[^>]*>([\s\S]*?)<\/a:p>/g;
      const tRegex = /<a:t\b[^>]*>([\s\S]*?)<\/a:t>/g;
      let m: RegExpExecArray | null;
      while ((m = pRegex.exec(xml)) !== null) {
        const inner = m[1];
        const parts: string[] = [];
        let tm: RegExpExecArray | null;
        while ((tm = tRegex.exec(inner)) !== null) {
          parts.push(decodeXmlEntities(tm[1]));
        }
        paragraphs.push(parts.join(""));
      }
      return paragraphs;
    }

    function decodeXmlEntities(s: string): string {
      return s.replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&quot;/g, "\"").replace(/&#39;/g, "'");
    }

    export async function parsePptxBuffer(input: ParseInput): Promise<DocumentIR> {
      let JSZip: any;
      try {
        JSZip = require("jszip");
      } catch {
        throw new Error("[E_DOC_DEP_MISSING] jszip not available. Ensure mammoth / pdfjs-dist deps are installed.");
      }
      const zip = await JSZip.loadAsync(input.buffer);

      // Enumerate slide files in order
      const slideFiles = Object.keys(zip.files)
        .filter((p) => /^ppt\/slides\/slide(\d+)\.xml$/.test(p))
        .sort((a, b) => {
          const na = parseInt(a.match(/slide(\d+)\.xml$/)![1], 10);
          const nb = parseInt(b.match(/slide(\d+)\.xml$/)![1], 10);
          return na - nb;
        });

      const body: SlideNode[] = [];
      const outline: OutlineItem[] = [];

      for (let i = 0; i < slideFiles.length; i++) {
        const slidePath = slideFiles[i];
        const slideIdx = i + 1;
        const slideXml: string = await zip.file(slidePath)!.async("string");
        const slideParas = extractPptxParagraphs(slideXml);

        const slideBody: ParagraphNode[] = slideParas
          .filter((t) => t.trim().length > 0)
          .map<ParagraphNode>((t) => ({ kind: "paragraph", runs: [{ text: t }], locator: { slide: slideIdx } }));

        // Notes
        const notesPath = `ppt/notesSlides/notesSlide${slideIdx}.xml`;
        const notesFile = zip.file(notesPath);
        let notes: ParagraphNode[] | undefined;
        if (notesFile) {
          const notesXml = await notesFile.async("string");
          const notesParas = extractPptxParagraphs(notesXml).filter((t) => t.trim().length > 0);
          if (notesParas.length > 0) {
            notes = notesParas.map<ParagraphNode>((t) => ({ kind: "paragraph", runs: [{ text: t }], locator: { slide: slideIdx } }));
          }
        }

        const title = slideParas.find((t) => t.trim().length > 0)?.trim() ?? `Slide ${slideIdx}`;
        body.push({
          kind: "slide",
          index: slideIdx,
          title,
          body: slideBody,
          notes,
          locator: { slide: slideIdx },
        });
        outline.push({ level: 1, title, locator: { slide: slideIdx } });
      }

      return {
        source: { path: input.path, format: "pptx", bytes: input.buffer.length, sha256: input.sha256 },
        meta: { pages: slideFiles.length },
        outline,
        body,
        media: [],
      };
    }

    export const pptxParser: DocumentParser = { format: "pptx", parse: parsePptxBuffer };
    ```
